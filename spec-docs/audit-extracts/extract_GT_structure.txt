    49	const ordinalSuffix = (num: number) => {
    50	  if (num % 100 >= 11 && num % 100 <= 13) return "th";
    51	  switch (num % 10) {
    52	    case 1:
    53	      return "st";
    54	    case 2:
    55	      return "nd";
    56	    case 3:
    57	      return "rd";
    58	    default:
    59	      return "th";
    60	  }
    61	};
